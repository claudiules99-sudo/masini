// Parser ICS minimal cu expandare RRULE (WEEKLY/MONTHLY/YEARLY, INTERVAL, UNTIL, COUNT, BYDAY simplu),
// EXDATE, RECURRENCE-ID și STATUS:CANCELLED. Lucrează pe ore locale (Europe/Bucharest) ca șiruri.
// Folosit identic în Apps Script și în teste.
function icsEvents(text, fromYmd, toYmd, zToLocal) {
  var lines = text.replace(/\r?\n[ \t]/g, '').split(/\r?\n/);
  var evs = [], cur = null;
  for (var i = 0; i < lines.length; i++) {
    var l = lines[i];
    if (l === 'BEGIN:VEVENT') { cur = {}; continue; }
    if (l === 'END:VEVENT') { if (cur) evs.push(cur); cur = null; continue; }
    if (!cur) continue;
    var c = l.indexOf(':'); if (c < 0) continue;
    var head = l.slice(0, c), val = l.slice(c + 1), name = head.split(';')[0];
    if (name === 'EXDATE') { (cur.ex = cur.ex || []).push.apply(cur.ex, val.split(',').map(function (v) { return norm(v, head); })); }
    else if (name === 'DTSTART' || name === 'DTEND' || name === 'RECURRENCE-ID') cur[name] = norm(val, head);
    else if (name === 'SUMMARY') cur.SUMMARY = val.replace(/\\,/g, ',').replace(/\\;/g, ';').replace(/\\n/g, ' ').replace(/\\\\/g, '\\');
    else if (name === 'RRULE' || name === 'UID' || name === 'STATUS') cur[name] = val;
  }
  function norm(v, head) { // -> 'YYYYMMDDTHHMM' local
    v = v.trim();
    if (/Z$/.test(v)) return zToLocal(v);
    if (/^\d{8}$/.test(v)) return v + 'T0000';
    return v.slice(0, 13);
  }
  var lo = fromYmd.replace(/-/g, '') + 'T0000', hi = toYmd.replace(/-/g, '') + 'T0000';
  var overrides = {}, out = [];
  evs.forEach(function (e) { if (e['RECURRENCE-ID']) overrides[e.UID + '|' + e['RECURRENCE-ID']] = e; });
  function push(e, start) {
    if (start < lo || start >= hi) return;
    if (e.STATUS === 'CANCELLED') return;
    out.push({ uid: e.UID, start: start, summary: (e.SUMMARY || '').trim() });
  }
  evs.forEach(function (e) {
    if (!e.DTSTART || e['RECURRENCE-ID']) { if (e.DTSTART && e['RECURRENCE-ID']) push(e, e.DTSTART); return; }
    if (!e.RRULE) { push(e, e.DTSTART); return; }
    var r = {}; e.RRULE.split(';').forEach(function (p) { var kv = p.split('='); r[kv[0]] = kv[1]; });
    var until = r.UNTIL ? (/Z$/.test(r.UNTIL) ? zToLocal(r.UNTIL) : (r.UNTIL.length === 8 ? r.UNTIL + 'T2359' : r.UNTIL.slice(0, 13))) : null;
    var count = r.COUNT ? +r.COUNT : null, interval = +(r.INTERVAL || 1);
    var ex = {}; (e.ex || []).forEach(function (x) { ex[x] = 1; });
    var t = e.DTSTART.slice(9), d = ymdToDate(e.DTSTART), n = 0, guard = 0;
    var days = r.BYDAY && r.FREQ === 'WEEKLY' ? r.BYDAY.split(',').map(function (s) { return ['SU','MO','TU','WE','TH','FR','SA'].indexOf(s.slice(-2)); }) : null;
    while (guard++ < 5000) {
      var cands = [];
      if (days) { var wk = new Date(d); wk.setDate(wk.getDate() - wk.getDay()); days.forEach(function (wd) { var x = new Date(wk); x.setDate(x.getDate() + wd); if (x >= ymdToDate(e.DTSTART)) cands.push(x); }); cands.sort(function (a, b) { return a - b; }); }
      else cands = [new Date(d)];
      var stop = false;
      for (var k = 0; k < cands.length; k++) {
        var s = dateToYmd(cands[k]) + t;
        if (until && s > until) { stop = true; break; }
        if (count && n >= count) { stop = true; break; }
        n++;
        if (s >= hi) { stop = true; break; }
        if (ex[s]) continue;
        var ov = overrides[e.UID + '|' + s];
        if (ov) continue; // instanța modificată e tratată separat
        push(e, s);
      }
      if (stop) break;
      if (r.FREQ === 'WEEKLY') d.setDate(d.getDate() + 7 * interval);
      else if (r.FREQ === 'DAILY') d.setDate(d.getDate() + interval);
      else if (r.FREQ === 'MONTHLY') d.setMonth(d.getMonth() + interval);
      else if (r.FREQ === 'YEARLY') d.setFullYear(d.getFullYear() + interval);
      else break;
    }
  });
  out.sort(function (a, b) { return a.start < b.start ? -1 : 1; });
  return out.map(function (o) { return { date: o.start.slice(0, 4) + '-' + o.start.slice(4, 6) + '-' + o.start.slice(6, 8), time: o.start.slice(9, 11) + ':' + o.start.slice(11, 13), summary: o.summary }; });
  function ymdToDate(s) { return new Date(+s.slice(0, 4), +s.slice(4, 6) - 1, +s.slice(6, 8)); }
  function dateToYmd(x) { return '' + x.getFullYear() + ('0' + (x.getMonth() + 1)).slice(-2) + ('0' + x.getDate()).slice(-2) + 'T'; }
}
/**
 * Hannah — backend pentru aplicația lui Hannah (pe contul Google al lui Claudiu, proiect separat).
 * Ține datele aplicației (sincronizate între dispozitive) și citește calendarul ei din iCloud.
 *
 * Proprietăți (Project Settings → Script properties):
 *   TOKEN    cheia secretă a aplicației (se generează cu setup())
 *   ICS_URL  linkul public iCloud al calendarului cu orele lui Hannah (webcal://... sau https://...)
 */

var TZ = 'Europe/Bucharest';
var CHUNK = 8000;

function prop_(k) { return PropertiesService.getScriptProperties().getProperty(k) || ''; }
function json_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }
function auth_(t) { var tok = prop_('TOKEN'); return tok && t === tok; }

/* ---------- stocare documente ---------- */
function readAll_() {
  var all = PropertiesService.getScriptProperties().getProperties(), docs = {}, parts = {};
  Object.keys(all).forEach(function (k) {
    if (k.indexOf('d:') !== 0) return;
    var m = k.match(/^d:(.+?)(?:#(\d+))?$/);
    if (m[2] === undefined) { if (all[k] !== '__chunked__') docs[m[1]] = all[k]; }
    else (parts[m[1]] = parts[m[1]] || [])[+m[2]] = all[k];
  });
  Object.keys(parts).forEach(function (p) { docs[p] = parts[p].join(''); });
  var out = {};
  Object.keys(docs).forEach(function (p) { try { out[p] = JSON.parse(docs[p]); } catch (e) {} });
  return out;
}
function writeDoc_(path, data) {
  var sp = PropertiesService.getScriptProperties();
  deleteDoc_(path);
  var s = JSON.stringify(data);
  if (s.length <= CHUNK) { sp.setProperty('d:' + path, s); return; }
  var o = {}; o['d:' + path] = '__chunked__';
  for (var i = 0, n = 0; i < s.length; i += CHUNK, n++) o['d:' + path + '#' + n] = s.slice(i, i + CHUNK);
  sp.setProperties(o);
}
function deleteDoc_(path) {
  var sp = PropertiesService.getScriptProperties();
  var keys = sp.getKeys();
  keys.forEach(function (k) { if (k === 'd:' + path || k.indexOf('d:' + path + '#') === 0) sp.deleteProperty(k); });
}

/* ---------- HTTP ---------- */
function doGet(e) {
  var p = e.parameter || {};
  if (!auth_(p.t)) return json_({ error: 'unauthorized' });
  try {
    if (p.a === 'all') return json_({ docs: readAll_(), at: Date.now() });
    if (p.a === 'cal') return json_({ events: oreEvents_(p.from, p.to) });
    if (p.a === 'ping') return json_({ ok: true });
    if (p.a === 'vapid') return json_({ key: vapidKeys_().pub });
    if (p.a === 'pending') return json_(pendingToday_());
    return json_({ error: 'unknown action' });
  } catch (err) { return json_({ error: String(err) }); }
}
function doPost(e) {
  var b; try { b = JSON.parse(e.postData.contents); } catch (x) { return json_({ error: 'bad json' }); }
  if (!auth_(b.t)) return json_({ error: 'unauthorized' });
  var lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    if (b.a === 'set') { writeDoc_(b.p, b.d); return json_({ ok: true }); }
    if (b.a === 'del') { deleteDoc_(b.p); return json_({ ok: true }); }
    if (b.a === 'sub') return json_({ ok: true, n: addSub_(b.sub) });
    if (b.a === 'bulk') { (b.ops || []).forEach(function (o) { if (o.d === null) deleteDoc_(o.p); else writeDoc_(o.p, o.d); }); return json_({ ok: true }); }
    return json_({ error: 'unknown action' });
  } catch (err) { return json_({ error: String(err) }); }
  finally { lock.releaseLock(); }
}

/* ---------- calendar iCloud ---------- */
function zToLocal_(v) {
  var d = new Date(Date.UTC(+v.slice(0, 4), +v.slice(4, 6) - 1, +v.slice(6, 8), +v.slice(9, 11), +v.slice(11, 13)));
  return Utilities.formatDate(d, TZ, "yyyyMMdd'T'HHmm");
}
function oreEvents_(from, to) {
  var cache = CacheService.getScriptCache(), key = 'ore:' + from + ':' + to;
  var hit = cache.get(key); if (hit) return JSON.parse(hit);
  var url = prop_('ICS_URL').replace(/^webcal:/, 'https:');
  if (!url) return [];
  var r = UrlFetchApp.fetch(url, { muteHttpExceptions: true });
  if (r.getResponseCode() !== 200) throw new Error('iCloud a răspuns ' + r.getResponseCode());
  var evs = icsEvents(r.getContentText(), from, to, zToLocal_);
  try { cache.put(key, JSON.stringify(evs), 600); } catch (e) {}
  return evs;
}

  Logger.log('TOKEN: ' + prop_('TOKEN'));
}

// Curba P-256 implementată cu BigInt (Apps Script nu are ECDSA nativ).
var EC = (function () {
  var P = BigInt('0xffffffff00000001000000000000000000000000ffffffffffffffffffffffff');
  var N = BigInt('0xffffffff00000000ffffffffffffffffbce6faada7179e84f3b9cac2fc632551');
  var A = P - BigInt(3);
  var G = [BigInt('0x6b17d1f2e12c4247f8bce6e563a440f277037d812deb33a0f4a13945d898c296'), BigInt('0x4fe342e2fe1a7f9b8ee7eb4a7c0f9e162bce33576b315ececbb6406837bf51f5')];
  var Z = BigInt(0), ONE = BigInt(1), TWO = BigInt(2), THREE = BigInt(3);
  function mod(a, m) { var r = a % m; return r < Z ? r + m : r; }
  function inv(a, m) { var lm = ONE, hm = Z, low = mod(a, m), high = m; while (low > ONE) { var r = high / low; var nm = hm - lm * r, nw = high - low * r; hm = lm; high = low; lm = nm; low = nw; } return mod(lm, m); }
  // coordonate Jacobiene
  function dbl(p) { if (!p || p[1] === Z) return null; var X = p[0], Y = p[1], Zz = p[2];
    var YY = mod(Y * Y, P), S = mod(BigInt(4) * X * YY, P), ZZ = mod(Zz * Zz, P), M = mod(THREE * X * X + A * ZZ * ZZ, P);
    var X3 = mod(M * M - TWO * S, P), Y3 = mod(M * (S - X3) - BigInt(8) * YY * YY, P), Z3 = mod(TWO * Y * Zz, P); return [X3, Y3, Z3]; }
  function add(p, q) { if (!p) return q; if (!q) return p;
    var Z1Z1 = mod(p[2] * p[2], P), Z2Z2 = mod(q[2] * q[2], P), U1 = mod(p[0] * Z2Z2, P), U2 = mod(q[0] * Z1Z1, P);
    var S1 = mod(p[1] * q[2] * Z2Z2, P), S2 = mod(q[1] * p[2] * Z1Z1, P);
    if (U1 === U2) return S1 === S2 ? dbl(p) : null;
    var H = mod(U2 - U1, P), R = mod(S2 - S1, P), HH = mod(H * H, P), HHH = mod(H * HH, P), V = mod(U1 * HH, P);
    var X3 = mod(R * R - HHH - TWO * V, P), Y3 = mod(R * (V - X3) - S1 * HHH, P), Z3 = mod(p[2] * q[2] * H, P); return [X3, Y3, Z3]; }
  function mul(k, pt) { var r = null, q = [pt[0], pt[1], ONE]; while (k > Z) { if (k & ONE) r = add(r, q); q = dbl(q); k >>= ONE; } return r; }
  function affine(p) { var zi = inv(p[2], P), zi2 = mod(zi * zi, P); return [mod(p[0] * zi2, P), mod(p[1] * zi2 * zi, P)]; }
  function toInt(bytes) { var h = '0x'; for (var i = 0; i < bytes.length; i++) h += ('0' + (bytes[i] & 255).toString(16)).slice(-2); return BigInt(h); }
  function toBytes(x, len) { var h = x.toString(16); while (h.length < len * 2) h = '0' + h; var out = []; for (var i = 0; i < len; i++) out.push(parseInt(h.substr(i * 2, 2), 16)); return out; }
  return {
    N: N,
    pub: function (d) { var Q = affine(mul(d, G)); return [4].concat(toBytes(Q[0], 32), toBytes(Q[1], 32)); },
    // hash = 32 octeți; kFor(counter) -> 32 octeți (determinist, derivat din cheie+mesaj)
    sign: function (hash, d, kFor) {
      var z = toInt(hash);
      for (var c = 0; c < 50; c++) {
        var k = mod(toInt(kFor(c)), N - ONE) + ONE;
        var R = affine(mul(k, G)), r = mod(R[0], N); if (r === Z) continue;
        var s = mod(inv(k, N) * (z + r * d), N); if (s === Z) continue;
        return toBytes(r, 32).concat(toBytes(s, 32));
      }
      throw new Error('sign failed');
    },
    toInt: toInt, toBytes: toBytes
  };
})();

function s8_(a) { return a.map(function (b) { b = b & 255; return b > 127 ? b - 256 : b; }); }
function b64u_(bytes) { return Utilities.base64EncodeWebSafe(s8_(bytes)).replace(/=+$/, ''); }
function b64uStr_(s) { return Utilities.base64EncodeWebSafe(s, Utilities.Charset.UTF_8).replace(/=+$/, ''); }
function fromB64u_(s) { return Utilities.base64DecodeWebSafe(s + '==='.slice((s.length + 3) % 4)); }
function u8_(a) { return a.map(function (b) { return b & 255; }); }

function vapidKeys_() {
  var sp = PropertiesService.getScriptProperties();
  var d = sp.getProperty('VAPID_D'), pub = sp.getProperty('VAPID_PUB');
  if (d && pub) return { d: EC.toInt(u8_(fromB64u_(d))), pub: pub };
  var seed = []; for (var i = 0; i < 4; i++) seed = seed.concat(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, Utilities.getUuid() + Date.now() + Math.random()));
  var dn = EC.toInt(u8_(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, s8_(seed)))) % (EC.N - BigInt(1)) + BigInt(1);
  var dB = EC.toBytes(dn, 32), pB = EC.pub(dn);
  sp.setProperty('VAPID_D', b64u_(dB)); sp.setProperty('VAPID_PUB', b64u_(pB));
  return { d: dn, pub: b64u_(pB) };
}
function vapidJwt_(aud) {
  var keys = vapidKeys_();
  var head = b64uStr_(JSON.stringify({ typ: 'JWT', alg: 'ES256' }));
  var body = b64uStr_(JSON.stringify({ aud: aud, exp: Math.floor(Date.now() / 1000) + 12 * 3600, sub: 'https://claudiules99-sudo.github.io/masini/hannah/' }));
  var input = head + '.' + body;
  var hash = u8_(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, input, Utilities.Charset.UTF_8));
  var dB = EC.toBytes(keys.d, 32);
  var sig = EC.sign(hash, keys.d, function (c) { return u8_(Utilities.computeHmacSha256Signature(s8_(hash.concat([c])), s8_(dB))); });
  return { jwt: input + '.' + b64u_(sig), pub: keys.pub };
}
function subs_() { try { return JSON.parse(prop_('PUSH_SUBS') || '[]'); } catch (e) { return []; } }
function saveSubs_(l) { PropertiesService.getScriptProperties().setProperty('PUSH_SUBS', JSON.stringify(l)); }
function addSub_(sub) { if (!sub || !sub.endpoint) throw new Error('bad sub'); var l = subs_().filter(function (s) { return s.endpoint !== sub.endpoint; }); l.push({ endpoint: sub.endpoint, at: Date.now() }); saveSubs_(l); return l.length; }
function sendPushAll_() {
  var l = subs_(), keep = [], res = [];
  l.forEach(function (s) {
    var m = s.endpoint.match(/^https:\/\/[^/]+/); var v = vapidJwt_(m[0]);
    var r = UrlFetchApp.fetch(s.endpoint, { method: 'post', muteHttpExceptions: true, payload: '', headers: { TTL: '7200', Urgency: 'high', Authorization: 'vapid t=' + v.jwt + ', k=' + v.pub } });
    var code = r.getResponseCode(); res.push(code + ' ' + r.getContentText().slice(0, 120));
    if (code !== 404 && code !== 410) keep.push(s);
  });
  if (keep.length !== l.length) saveSubs_(keep);
  return res;
}

/* ore neconfirmate azi (pentru notificare) */
function splitKids_(s, groups) { var n = String(s).trim().replace(/\s+/g, ' '); var g = (groups || []).filter(function (x) { return n.toLowerCase().indexOf(x.toLowerCase()) === 0; })[0]; return g ? [g] : n.split(/\s+(?:si|și|\+|&)\s+/i); }
function pendingToday_() {
  var now = new Date(), today = Utilities.formatDate(now, TZ, 'yyyy-MM-dd'), tom = Utilities.formatDate(new Date(now.getTime() + 864e5), TZ, 'yyyy-MM-dd');
  var docs = readAll_(), cfg = docs['cfg/main'] || {}, ignore = cfg.ignore || [];
  var od = docs['ore/' + today.slice(0, 7)] || { d: {} }, done = (od.d || {})[today] || {};
  var kids = [];
  oreEvents_(today, tom).forEach(function (e) {
    if (ignore.indexOf(e.summary.trim().replace(/\s+/g, ' ')) >= 0) return;
    splitKids_(e.summary, cfg.groups || ['Timi si Sofi']).forEach(function (k) { if (!done[k] && kids.indexOf(k) < 0) kids.push(k); });
  });
  return { date: today, kids: kids };
}
/** Rulează zilnic la ~19:50: dacă azi sunt ore neconfirmate, trimite notificarea. */
function notifyLessons() {
  var p = pendingToday_();
  if (!p.kids.length) return;
  Logger.log(sendPushAll_().join(' | '));
}
/** Rulează o dată din editor: cheile VAPID + verificarea zilnică la 19:50. */
function setupPush() {
  vapidKeys_();
  ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === 'notifyLessons') ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('notifyLessons').timeBased().everyDays(1).atHour(19).nearMinute(50).inTimezone(TZ).create();
  Logger.log('VAPID_PUB: ' + prop_('VAPID_PUB') + ' | abonați: ' + subs_().length);
}
